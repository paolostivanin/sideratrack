cmake_minimum_required(VERSION 3.25)
project(SideraStack VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(SIDERASTACK_BUILD_GUI "Build the desktop application" ON)
option(SIDERASTACK_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(SIDERASTACK_SYSTEM_SEP "Use a distribution-provided SEP library" OFF)
find_package(PkgConfig REQUIRED)
foreach(dep cfitsio libxml-2.0 liblz4 libzstd sqlite3 rtprocess)
  string(MAKE_C_IDENTIFIER ${dep} ident)
  pkg_check_modules(${ident} REQUIRED IMPORTED_TARGET ${dep})
endforeach()
find_package(Qt6 6.5 REQUIRED COMPONENTS Core)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(TBB REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(ZLIB REQUIRED)
if(SIDERASTACK_SYSTEM_SEP)
  pkg_check_modules(SEP REQUIRED IMPORTED_TARGET sep)
  add_library(sep ALIAS PkgConfig::SEP)
else()
  add_library(sep STATIC vendor/sep/src/analyse.c vendor/sep/src/aperture.c
    vendor/sep/src/background.c vendor/sep/src/convolve.c vendor/sep/src/deblend.c
    vendor/sep/src/extract.c vendor/sep/src/lutz.c vendor/sep/src/util.c)
  target_include_directories(sep PUBLIC vendor/sep/src)
  target_compile_definitions(sep PRIVATE SEP_VERSION_STRING="1.4.1")
  target_link_libraries(sep PUBLIC m)
endif()
add_library(siderastack_core STATIC src/io.cpp src/project.cpp src/processing.cpp
  src/pipeline.cpp src/xisf/common.cpp src/xisf/source.cpp src/xisf/metadata.cpp src/xisf/fits.cpp)
target_include_directories(siderastack_core PUBLIC include PRIVATE src/xisf)
target_link_libraries(siderastack_core PUBLIC Qt6::Core PkgConfig::cfitsio
  PkgConfig::libxml_2_0 PkgConfig::liblz4 PkgConfig::libzstd PkgConfig::sqlite3
  PkgConfig::rtprocess OpenSSL::Crypto ZLIB::ZLIB TBB::tbb Eigen3::Eigen sep)
target_compile_definitions(siderastack_core PUBLIC SIDERASTACK_VERSION="${PROJECT_VERSION}")
target_compile_definitions(siderastack_core PUBLIC QT_NO_KEYWORDS)
if(NOT SIDERASTACK_SYSTEM_SEP)
  target_compile_definitions(siderastack_core PRIVATE SIDERASTACK_THREADSAFE_SEP)
endif()
target_compile_options(siderastack_core PRIVATE -Wall -Wextra -Wpedantic -Wformat=2 -ffp-contract=off)
if(SIDERASTACK_SANITIZERS)
  if(NOT SIDERASTACK_SYSTEM_SEP)
    target_compile_options(sep PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
  target_compile_options(siderastack_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(siderastack_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(siderastack-cli src/cli.cpp)
target_link_libraries(siderastack-cli PRIVATE siderastack_core)
include(GNUInstallDirs)
install(TARGETS siderastack-cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(SIDERASTACK_BUILD_GUI)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
  add_executable(siderastack src/gui.cpp)
  target_link_libraries(siderastack PRIVATE siderastack_core Qt6::Widgets)
  install(TARGETS siderastack RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES data/org.siderastack.SideraStack.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES data/org.siderastack.SideraStack.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endif()
install(FILES LICENSE README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/siderastack)
install(DIRECTORY vendor/sep/licenses DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/siderastack/sep)
install(FILES src/xisf/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/siderastack/xisf)
include(CTest)
if(BUILD_TESTING)
  add_executable(core-tests tests/core.cpp)
  target_link_libraries(core-tests PRIVATE siderastack_core)
  add_test(NAME core COMMAND core-tests)
  if(SIDERASTACK_BUILD_GUI)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
    add_executable(gui-tests tests/gui.cpp)
    target_link_libraries(gui-tests PRIVATE siderastack_core Qt6::Widgets Qt6::Test)
    add_test(NAME gui-scale COMMAND gui-tests)
    set_tests_properties(gui-scale PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
  endif()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    execute_process(COMMAND ${Python3_EXECUTABLE} -c "import numpy,astropy,lz4.block,zstandard" RESULT_VARIABLE no_python_deps OUTPUT_QUIET ERROR_QUIET)
    if(NOT no_python_deps)
      add_test(NAME independent-formats COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:siderastack-cli>)
    endif()
  endif()
endif()
